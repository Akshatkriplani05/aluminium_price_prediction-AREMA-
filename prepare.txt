3. Fix the TIME SERIES  
# --- Ensure we have a proper Series with DatetimeIndex and a freq ---
# Use the Date column as index (if not already)
series = data.set_index('Date')['Price'].sort_index()

# If the index has no freq, make it regular (daily) by reindexing/resampling.
# Use asfreq if the data is already daily but missing .freq; otherwise resample + interpolate.
if series.index.freq is None:
    # create a full daily index from min to max
    full_idx = pd.date_range(start=series.index.min(), end=series.index.max(), freq='D')
    series = series.reindex(full_idx)            # introduce NaNs where days missing
    series = series.interpolate(method='time')   # fill missing days
    series.index.name = 'Date'

# now series should be a proper daily-indexed Series with freq='D'
print("Series index freq:", series.index.freq)

4. Fit the ARIMA MODEL
# --- Fit ARIMA (keep the order you already used, e.g. (1,1,1)) ---
from statsmodels.tsa.arima.model import ARIMA
model = ARIMA(series, order=(1, 1, 0))   # change order if you want
model_fit = model.fit()
print(model_fit.summary())

5. FORECAST For next N days
# --- Forecast for the next n days and attach correct datetime index ---
n_forecast = 300
pred = model_fit.get_forecast(steps=n_forecast)
pred_mean = pred.predicted_mean
pred_ci = pred.conf_int()

# Build a future daily index starting the day after the last observed date
start_date = series.index[-1] + pd.Timedelta(days=1)
future_index = pd.date_range(start=start_date, periods=n_forecast, freq='D')

# Convert to Series/DataFrame with the proper index
pred_mean = pd.Series(pred_mean, index=future_index, name='predicted_mean')
pred_ci.index = future_index

# --- Plot observed + forecast with CI ---
import matplotlib.pyplot as plt

plt.figure(figsize=(12,5))
plt.plot(series, label='Original')
plt.plot(pred_mean, label='Forecast', color='orange')
plt.fill_between(future_index,
                 pred_ci.iloc[:, 0],
                 pred_ci.iloc[:, 1],
                 color='orange', alpha=0.2)
plt.legend()
plt.xlabel('Date')
plt.ylabel('Price')
plt.title('Original and Forecast (ARIMA(1,1,1))')
plt.show()
